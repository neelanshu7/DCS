def Breadth_First_Search(graph, start, goal):
    queue = deque([start])
    visited = set()
    visited.add(start.state)
    
    while queue:
        current = queue.popleft()
        
        if current.state == goal.state:
            print_solution(current)
            return
        
        for neighbor in graph[current]:
            if neighbor.state not in visited:
                visited.add(neighbor.state)
                neighbor.parent = current
                neighbor.path_cost = current.path_cost + 1
                queue.append(neighbor)
    
    print("No Solution")

def generate_neighbors(state):
    neighbors = []
    n = int(len(state)**0.5)
    zero_index = state.index(0)
    row, col = zero_index // n, zero_index % n
    
    moves = [(-1, 0), (1, 0), (0, -1), (0, 1)]
    
    for dr, dc in moves:
        new_row, new_col = row + dr, col + dc
        if 0 <= new_row < n and 0 <= new_col < n:
            new_index = new_row * n + new_col
            new_state = list(state)
            new_state[zero_index], new_state[new_index] = new_state[new_index], new_state[zero_index]
            neighbors.append(tuple(new_state))
    
    return neighbors

        return node['path_cost'] + h(node['state'])

            child = child_node(node, action, child_state, child_path_cost)

import heapq
def is_safe(board, row, col):
    """Check if placing a queen at (row, col) is safe"""
    # Check column
    for i in range(row):
        if board[i][col] == 1:
            return False
    
    # Check upper left diagonal
    for i, j in zip(range(row-1, -1, -1), range(col-1, -1, -1)):
        if board[i][j] == 1:
            return False
    
    # Check upper right diagonal
    for i, j in zip(range(row-1, -1, -1), range(col+1, 4)):
        if board[i][j] == 1:
            return False
    
    return True

def count_non_attacking_pairs(board, current_queens):
    """Heuristic: Count number of non-attacking pairs of queens"""
    if current_queens < 2:
        return 0  # Need at least 2 queens to form pairs
    
    non_attacking = 0
    positions = []
    
    # Get all queen positions
    for i in range(4):
        for j in range(4):
            if board[i][j] == 1:
                positions.append((i, j))
    
    # Check all pairs
    for i in range(len(positions)):
        for j in range(i+1, len(positions)):
            row1, col1 = positions[i]
            row2, col2 = positions[j]
            
            # Check if not attacking each other
            if row1 != row2 and col1 != col2 and abs(row1-row2) != abs(col1-col2):
                non_attacking += 1
    
    return non_attacking

def greedy_best_first_search():
    # Priority queue: (-non_attacking_pairs, row, board, queens_placed)
    # Using negative because heapq is min-heap and we want to maximize
    heap = []
    
    # Initialize with empty board
    initial_board = [[0 for _ in range(4)] for _ in range(4)]
    heapq.heappush(heap, (0, 0, initial_board, 0))
    
    while heap:
        _, row, current_board, queens_placed = heapq.heappop(heap)
        
        # If all queens are placed, return solution
        if queens_placed == 4:
            return current_board
        
        # Try placing queen in each column of current row
        for col in range(4):
            if is_safe(current_board, row, col):
                new_board = [r[:] for r in current_board]
                new_board[row][col] = 1
                new_queens = queens_placed + 1
                non_attacking = count_non_attacking_pairs(new_board, new_queens)
                
                # Push with negative heuristic to simulate max-heap
                heapq.heappush(heap, (-non_attacking, row+1, new_board, new_queens))
    
    return None  # No solution found (though 4-queens has solutions)

def print_solution(board):
    for row in board:
        print(' '.join('Q' if cell == 1 else '.' for cell in row))

import sys
def iterative_deepening_search(problem):
    for depth in range(sys.maxsize):
        result = depth_limited_search(problem,depth)
        if result!='cutoff':
            return result

def depth_limited_search(problem,limit):
    return recursive_dls(Node(problem.initial_state),problem,limit)

def recursive_dls(node,problem,limit):
    if problem.goal_test(node.state):
        return node
    elif limit==0:
        return 'cutoff'
    else:
        cutoff_occured = False
        for action in problem.actions(node.state):
            child = Node(problem.result(node.state,action),node,action,node.depth+1)
            result = recursive_dls(child,problem,limit-1)
            if result=='cutoff':
                cutoff_occured = True
            elif result is not None:
                return result
        if cutoff_occured:
            return 'cutoff'
        else:
            return None

def find_empty_tile(state):
    for i in range(3):
        for j in range(3):
            if state[i][j]==0:
                return i,j

def construct_state(curr_state,action):
    i,j = find_empty_tile(curr_state)
    if (i==0 and action=='up') or (i==2 and action=='down') or (j==0 and action=='left') or (j==2 and action=='right'):
        return None
    else:
        new_state = list(map(list,curr_state))
        if action=='up':
            new_state[i][j],new_state[i-1][j] = new_state[i-1][j],new_state[i][j]
        elif action=='down':
            new_state[i][j],new_state[i+1][j] = new_state[i+1][j],new_state[i][j]
        elif action=='left':
            new_state[i][j],new_state[i][j-1] = new_state[i][j-1],new_state[i][j]
        elif action=='right':
            new_state[i][j],new_state[i-1][j+1] = new_state[i-1][j+1],new_state[i][j]
        return list(map(list,new_state))

import random
def fitness_function(chromosome):
    conflicts = 0
    for i in range(7):
        for j in range(i+1,8):
            if chromosome[i]==chromosome[j] or abs(chromosome[i]-chromosome[j])==j-i:
                conflicts += 1
    return conflicts

def random_selection(population):
    total_fitness = sum(1/(fitness_function(chromosome)+1) for chromosome in population)
    rand_num = random.uniform(0,total_fitness)
    cumulative_fitness = 0
    for chromosome in population:
        cumulative_fitness += 1/(fitness_function(chromosome)+1)
        if cumulative_fitness >= rand_num:
            return chromosome

def reproduce(x,y):
    n = len(x)
    c = random.randint(1,n-1)
    child1 = x[:c]+y[c:]
    child2 = y[:c]+x[c:]
    return child1,child2

def mutate(child):
    n = len(child)
    c = random.randint(0,n-1)
    m = random.randint(1,8)
    child[c] = m
    return child

def genetic_algorithm(chromosome):
    population_size = 100
    max_generations = 1000
    mutation_rate = 0.1
    population = [chromosome.copy() for _ in range(population_size)]
    for generation in range(max_generations):
        new_population = []
        for _ in range(population_size // 2):
            parent1 = random_selection(population)
            parent2 = random_selection(population)
            child1,child2 = reproduce(parent1,parent2)
            if random.random() < mutation_rate:
                child1 = mutate(child1)
            if random.random() < mutation_rate:
                child2 = mutate(child2)
            new_population.append(child1)
            new_population.append(child2)
        population = new_population
        best_chromosome = min(population,key = lambda x:fitness_function(x))
        if fitness_function(best_chromosome) == 0:
            return best_chromosome,generation+1
    return None,max_generations

function GENETIC-ALGORITHM (POPULATION) returns an individual
 inputs: POPULATION, a set of individuals given as chromosomes along with their fitness function values to be calculated
  for i = 1 to POPULATION.size do
    POPULATION[i].Fitness_Value = FITNESS-FUNCTION (POPULATION[I].Chromosome) 
    repeat
           NEW-POPULATION ←empty set
           for i = 1 to (SIZE(population) / 2)  do
                   x ←RANDOM-SELECTION (POPULATION)
                   y ←RANDOM-SELECTION(POPULATION)
                   CHILD1 , CHILD2←REPRODUCE(x , y)
                  if (small random probability) then CHILD1 ←MUTATE(child )
                  if (small random probability) then CHILD2 ←MUTATE(child )
                  add CHILD1 & CHILD2 to NEW-POPULATION
           POPULATION ← NEW-POPULATION
       until some individual is fit enough, or enough time has elapsed
return the best individual in population, according to FITNESS-FN and also the number of iterations

function FITNESS-FUNCTION(x) returns a fitness function value
   input: x is a chromosome in the POPULATION
      FITNESSVALUE = 0
      for i = 1 to 7 
          for j = i+1 to 8
               if ( x[i] ≠ x[j] and ( | x[j] – x[i] | ≠ j – I )
                    FITNESSVALUE = FITNESSVALUE + 1
      return FITNESSVALUE

function RANDOM-SELECTION (POPULATION) returns an individual
 input: POPULATION contains possible candidates for the reproduction
 Select POPULATION[ i ] with probability proportional to POPULATION[ i ].Fitness_Value

      C2 = APPEND(SUBSTRING(y, 1, c), SUBSTRING(x, c + 1, n))
                  return  C1 and C2
