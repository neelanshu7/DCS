def dfsSearch(graph,start,goal):
    visited=[]
    stack=[start]
    while stack:
        node=stack.pop()
        if node not in visited:
            visited.append(node)
            stack.extend(reversed(graph[node]))
            if node==goal:
              print("Goal Node Found:",node)
              break
    print("DFS Traversal:",visited)

##### recursive version of DFS
def dfs_recursive(graph, node, goal, visited=None):
    if visited is None:
        visited = []

    visited.append(node)

    if node == goal:
        print("Goal Node Found:", node)
        print("DFS Traversal:", visited)
        return True

    for neighbor in graph[node]:
        if neighbor not in visited:
            if dfs_recursive(graph, neighbor, goal, visited):
                return True

    return False

def dls(graph,start,goal,limit):
    def rec_dls(node,goal,limit,path,visited):
        if node==goal:
            return path
        if limit<=0:
            return None
        visited.add(node)
        for neighbor in graph.get(node,[]):
            if neighbor not in visited:
                result=rec_dls(neighbor,goal,limit-1,path+[neighbor],visited)
                if result is not None:
                    return result
        return None
    return rec_dls(start,goal,limit,[start],set())

    def expand_layer(queue,visited,other_visited):
    current=queue.popleft()
    for neighbor in graph[current]:
        if neighbor not in visited:
            visited[neighbor]=current
            queue.append(neighbor)
            if neighbor in other_visited:
                return neighbor # Meeting point
    return None

    def build_path(meeting_point,front_visited,back_visited):
    path=[]
    node=meeting_point
    while node:
        path.append(node)
        node=front_visited[node]
    path=path[::-1]

    node=back_visited[meeting_point]
    while node:
        path.append(node)
        node=back_visited[node]
    return path

# Uniform Cost Search
import heapq

def uniform_cost_search(graph,start,goal):
    queue=[(0,start,[start])]
    visited=set()

    while queue:
        cost,current,path=heapq.heappop(queue)
        if current==goal:
            return path,cost
        if current in visited:
            continue
        visited.add(current)

        for neighbor,edge_cost in graph.get(current,[]):
            if neighbor not in visited:
                heapq.heappush(queue,(cost+edge_cost,neighbor,path+[neighbor]))
    return None, float('inf') # No path found

import random
N=8 # 8-Queens
POP_SIZE=50
MUTATION_RATE=0.1
GENERATIONS=1000
# Chromosome: list of length N, each value=row position for queen in the column
def random_chromosome():
    return [random.randint(0,N-1) for _ in range(N)]

# Fitness function: count non-attacking pairs
def fitness(chromosome):
    non_attacking = 0  
    total_pairs = (N * (N - 1)) // 2 # For N=8, this is 28  

    for i in range(N):
        for j in range(i + 1, N):
            # Queens attack if on same row or diagonal.
            if chromosome[i] != chromosome[j] and abs(chromosome[i] - chromosome[j]) != abs(i - j):
                non_attacking += 1  

    return non_attacking # Max = 28 for N=8

# Selection: tournament
def selection(population):
    tournament_size = 5
    tournament = random.sample(population, tournament_size)
    tournament.sort(key=lambda ch: fitness(ch), reverse=True)
    return tournament[0]

# Crossover: single point
def crossover(parent1, parent2):
    point = random.randint(1, N - 2)
    return parent1[:point] + parent2[point:]

# Mutation: change row of a queen randomly
def mutate(chromosome):
    for i in range(N):
        if random.random() < MUTATION_RATE:
            chromosome[i] = random.randint(0, N - 1)
    return chromosome

# Main GA Loop
population = [random_chromosome() for _ in range(POP_SIZE)]

for generation in range(GENERATIONS):
    population.sort(key=lambda ch: fitness(ch), reverse=True)

    # Print best solution of this generation
    best = population[0]
    best_fit = fitness(best)
    print(f"Gen {generation}: Best = {best}, Fitness = {best_fit}")

    # Stop if solution found
    if best_fit == (N * (N - 1)) // 2:
        print("Solution found!")
        break

    # Create new population
    new_population = population[:2] # Elitism: carry best 2
    while len(new_population) < POP_SIZE:
        parent1 = selection(population)
        parent2 = selection(population)
        child = crossover(parent1, parent2)
        child = mutate(child)
        new_population.append(child)

    population = new_population
