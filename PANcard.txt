7. Alpha Beta Pruning
import math

def alpha_beta_pruning(depth, node_index, maximizing_player, values, alpha, beta, path=None):
    if path is None:
        path = []

    # Base case: leaf node
    if depth == 0 or node_index >= len(values):
        return values[node_index], node_index, path + [node_index]

    if maximizing_player:
        max_eval = -math.inf
        best_move = None
        best_path = None

        for i in range(2): 
            eval, move, new_path = alpha_beta_pruning(
                depth - 1, node_index * 2 + i, False, values, alpha, beta, path + [node_index])

            if eval > max_eval:
                max_eval = eval
                best_move = move
                best_path = new_path  # Corrected to new_path

            alpha = max(alpha, eval)
            if beta <= alpha:
                break  

        return max_eval, best_move, best_path

    else: 
        min_eval = math.inf
        best_move = None
        best_path = None

        path = path + [node_index]

        for i in range(2):  
            eval, move, new_path = alpha_beta_pruning(
                depth - 1, node_index * 2 + i, True, values, alpha, beta, path)

            if eval < min_eval:
                min_eval = eval
                best_move = move
                best_path = new_path  # Corrected to new_path

            beta = min(beta, eval)
            if beta <= alpha:
                break

        return min_eval, best_move, best_path

from copy import deepcopy

regions = ["WA", "NT", "SA", "Q", "NSW", "V", "I"]
colors = ["Red", "Green", "Blue"]

neighbors = {
    "WA": ["NT", "SA"],
    "NT": ["WA", "SA", "Q"],
    "SA": ["WA", "NT", "Q", "NSW", "V"],
    "Q": ["NT", "SA", "NSW"],
    "NSW": ["SA", "Q", "V"],
    "V": ["SA", "NSW"],
    "I": []
}

domains_init = {r: list(colors) for r in regions}

def select_unassigned_variable(assignment, domains):
    unassigned = [v for v in regions if v not in assignment]

    # Fix: Use correct variable
    domain_sizes = {v: len(domains[v]) for v in unassigned}

    min_size = min(domain_sizes.values())
    mrv_candidates = [v for v, s in domain_sizes.items() if s == min_size]

    if len(mrv_candidates) == 1:
        return mrv_candidates[0], mrv_candidates, None
    else:
        deg_counts = {v: sum(1 for n in neighbors[v] if n not in assignment) for v in mrv_candidates}
        max_deg = max(deg_counts.values())
        degree_candidates = [v for v, d in deg_counts.items() if d == max_deg]

        return degree_candidates[0], mrv_candidates, deg_counts

def lcv_ordering(var, domains, assignment):
    scores = []

    for val in domains[var]:
        eliminated = 0
        for n in neighbors[var]:
            if n not in assignment and val in domains[n]:
                eliminated += 1
        scores.append((val, eliminated))

    # Sort scores before returning
    scores.sort(key=lambda x: x[1])
    return [v for v, _ in scores], scores

def forward_check(var, value, domains, assignment):
    removed = []

    for n in neighbors[var]:
        if n in assignment:
            if assignment[n] == value:
                return None

        if value in domains[n]:
            domains[n].remove(value)
            removed.append((n, value))
            if not domains[n]:
                return None

    return removed

steps = []

def backtrack(assignment, domains):
    if len(assignment) == len(regions):
        return assignment

    var, mrv_candidates, deg_counts = select_unassigned_variable(assignment, domains)
    ordered_vals, lcv_scores = lcv_ordering(var, domains, assignment)

    step_log = {
        "var": var,
        "mrv_candidates": mrv_candidates,
        "tie_info": None,  # Fix: renamed to avoid undefined variable
        "domains_before": deepcopy(domains),
        "lcv_scores": lcv_scores,
        "lcv_orders": ordered_vals  # Fix typo
    }

    steps.append(step_log)

    for val in ordered_vals:
        if any(assignment.get(n) == val for n in neighbors[var]):
            continue

        assignment[var] = val
        local_domains = deepcopy(domains)

        removed = forward_check(var, val, local_domains, assignment)

        if removed is not None:
            result = backtrack(assignment, local_domains)
            if result:
                return result

        del assignment[var]

    return None

solution = backtrack({}, deepcopy(domains_init))

print("\n=== Australia Map Coloring (Forward Checking + MRV + Degree + LCV) ===\n")

if solution:
    print("Solution found:")
    for r in sorted(solution):
        print(f" {r}: {solution[r]}")
else:
    print("No solution found.")

print("\n=== Search Trace ===\n")

for i, s in enumerate(steps, 1):
    print(f"Step {i}: Selected {s['var']}")
    print(f"  MRV candidates: {s['mrv_candidates']} "
          f"({[(v, len(s['domains_before'][v])) for v in s['mrv_candidates']]})")
    if s["tie_info"]:
        print(f"  Tie-Break (Degree): {s['tie_info']}")

from itertools import permutations

def word_to_number(word,mapping):
  return int(''.join([str(mapping[ch]) for ch in word]))

def is_valid(mapping, words,result):
  for w in words + [result]:
    if mapping[w[0]]==0:
      return False
  total = sum(word_to_number(w,mapping) for w in words)
  result_value = word_to_number(result,mapping)
  return total==result_value

def solve_cryptarithmetic(words,result):
  unique_letters = set("".join(words)+result)
  if len(unique_letters)>10:
    print("Too many letters (more than 10). Cannot solve.")
    return
  for perm in permutations(range(10),len(unique_letters)):
    mapping = dict(zip(unique_letters,perm))
    if is_valid(mapping,words,result):
      return mapping

  return None

words = ["SEND","MORE"]
result="MONEY"
solution = solve_cryptarithmetic(words,result)
if solution:
  print("Solution found:")
  for letter,digit in solution.items():
    print(f"{letter}={digit}")

  word_values = [word_to_number(w,solution) for w in words]
  result_value = word_to_number(result,solution)
  print(f"\n{words[0]}+{words[1]}={result}")
  print(f"{word_values[0]}+{word_values[1]}={result_value}")
else:
  print("No solution found.")
